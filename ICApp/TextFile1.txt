using HalconDotNet;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows;
using VisualSorting.Services;
using static System.Net.Mime.MediaTypeNames;
   
namespace VisualSorting.Services
{
    
    //模板服务实现类
  
    public class TemplateService : ITemplateService
    {
        #region 私有字段
       
        // 模板区域（矩形ROI）
       
        private HObject ho_templateRegion = null;

      
        // 形状模板ID
       
        private HTuple hv_modelID = new HTuple();

      
        // 模板轮廓
      
        private HObject ho_modelContours = null;

        
        // 模板区域XLD轮廓
      
        private HObject ho_templateRegionXld = null;
        #endregion

        #region 属性实现
        
        // 模板是否已创建
        
        public bool IsTemplateCreated { get; private set; }

      
        // 模板区域
       
        public HObject TemplateRegion
        {
            get => ho_templateRegion;
            private set
            {
                // 释放旧区域，避免内存泄漏
                if (ho_templateRegion != null && ho_templateRegion.IsInitialized())
                {
                    ho_templateRegion.Dispose();
                }
                ho_templateRegion = value;

                // 验证新区域有效性
                if (ho_templateRegion != null && ho_templateRegion.IsInitialized())
                {
                    // 计算区域面积，面积太小则无效
                    HOperatorSet.AreaCenter(ho_templateRegion, out HTuple area, out _, out _);
                    if (area.D <= 100)
                    {
                        // 面积小于100像素，无效区域
                        ho_templateRegion.Dispose();
                        ho_templateRegion = null;
                        throw new Exception("生成的模板区域无效，请检查绘制的坐标范围");
                    }
                }
                else
                {
                    throw new Exception("生成的模板区域无效，区域为空");
                }
            }
        }

      
        // 模板区域XLD轮廓
        
        public HObject TemplateRegionXld { get => ho_templateRegionXld; set => ho_templateRegionXld = value; }

        /// <summary>
        /// 形状模板ID
        /// FindShapeModel使用此ID进行模板匹配
        /// </summary>
        public HTuple ModelID { get; private set; }

        /// <summary>
        /// 模板轮廓
        /// 用于显示匹配位置
        /// </summary>
        public HObject ModelContours { get; private set; }
        #endregion

        #region 构造函数
        /// <summary>
        /// 无参构造函数
        /// 模板服务不依赖其他服务，独立创建模板数据
        /// </summary>
        public TemplateService()
        {
            IsTemplateCreated = false;
            ModelID = null;
            ModelContours = null;
        }
        #endregion

        #region 公共方法
        /// <summary>
        /// 设置模板区域（不创建模板）
        /// 仅用于界面显示模板位置
        /// </summary>
        /// <param name="hv_mRow1">矩形左上角行坐标</param>
        /// <param name="hv_mColumn1">矩形左上角列坐标</param>
        /// <param name="hv_mRow2">矩形右下角行坐标</param>
        /// <param name="hv_mColumn2">矩形右下角列坐标</param>
        public void SetTemplateRegion(double hv_mRow1, double hv_mColumn1, double hv_mRow2, double hv_mColumn2)
        {
            // 释放旧的XLD轮廓
            if (TemplateRegionXld != null && TemplateRegionXld.IsInitialized())
            {
                TemplateRegionXld.Dispose();
                TemplateRegionXld = null;
            }

            try
            {
                // ========== 生成矩形区域 ==========
                // GenRectangle1：创建轴对齐矩形
                // 参数：左上角(row1,col1)，右下角(row2,col2)
                HOperatorSet.GenRectangle1(out HObject rext, hv_mRow1, hv_mColumn1, hv_mRow2, hv_mColumn2);
                TemplateRegion = rext;

                // ========== 转换为XLD轮廓 ==========
                // GenContourRegionXld：将区域转换为XLD轮廓
                // "border"参数表示只提取边界轮廓
                HOperatorSet.GenContourRegionXld(rext, out HObject templateXld, "border");

                if (templateXld == null || templateXld.IsInitialized() == false)
                {
                    throw new Exception("生成的模板区域无效，请检查绘制的坐标范围");
                }

                TemplateRegionXld = templateXld;
            }
            catch (Exception ex)
            {
                // 创建失败时清理资源
                ho_templateRegion?.Dispose();
                ho_templateRegion = null;
                TemplateRegionXld?.Dispose();
                TemplateRegionXld = null;
                throw new Exception($"设置模板区域失败：{ex.Message}");
            }
        }

     
        // 创建形状模板
        // 核心方法：从图像中提取形状特征，生成可匹配的模板
        // 模板创建流程：
        // 1. 锁定模板区域图像
        // 2. 提取形状特征
        // 3. 生成模板ID
        // 4. 提取模板轮廓
      
        // <param name="ho_GrayImage">灰度图像</param>
        // <param name="hv_mRow1">矩形区域左上角行坐标</param>
        // <param name="hv_mColumn1">矩形区域左上角列坐标</param>
        // <param name="hv_mRow2">矩形区域右下角行坐标</param>
        // <param name="hv_mColumn2">矩形区域右下角列坐标</param>
        public void CreateTemplate(HObject ho_GrayImage, double hv_mRow1, double hv_mColumn1, double hv_mRow2, double hv_mColumn2)
        {
            // ========== 输入验证 ==========
            if (ho_GrayImage == null || !ho_GrayImage.IsInitialized())
            {
                throw new Exception("输入图像无效");
            }
            if (TemplateRegionXld == null || !TemplateRegionXld.IsInitialized())
            {
                throw new Exception("请先设置有效的模板区域");
            }

            HObject ho_modelContours = null;
            HObject ho_ImageReduced = null;
            HObject ho_ImageMean = null;
            HObject ho_Edges = null;
            HObject ho_FinalXLD = null;
            HTuple hv_modelID = null;

            try
            {
                // ========== 步骤1：锁定模板区域 ==========
                // ReduceDomain：将图像定义域限制在模板区域内
                // 后续操作只在该区域内进行，提高效率
                HOperatorSet.ReduceDomain(ho_GrayImage, ho_templateRegion, out ho_ImageReduced);
                // 平滑图像：使用均值滤波减少噪声，增强边缘特征
                HOperatorSet.MeanImage(ho_ImageReduced, out ho_ImageMean, 9, 9);
                // 提取边缘：使用Canny算子提取亚像素级边缘，增强模板特征
                HOperatorSet.EdgesSubPix(ho_ImageMean, out ho_Edges, "canny", 1.5, 15, 30);
                // 选择边缘：根据轮廓长度过滤边缘，去除过短的噪声边缘，保留主要特征
                HOperatorSet.SelectShapeXld(ho_Edges, out ho_FinalXLD, "contlength", "and", 300, 99999);

                // ========== 步骤2：创建形状模板 ==========
                // CreateShapeModel：创建形状模板
                // 参数说明：
                // - "auto" 自动选择金字塔层级和角度步长
                // - (-360, 360) 允许360度旋转匹配
                // - "ignore_global_polarity" 忽略整体明暗变化
                HOperatorSet.CreateShapeModelXld(ho_FinalXLD, "auto", (new HTuple(-360)).TupleRad()
                 , (new HTuple(360)).TupleRad(), "auto", "auto", "ignore_local_polarity",  5, out hv_modelID);

                // 验证模板创建成功
                if (hv_modelID.Length == 0)
                {
                    throw new Exception("创建模板失败，ROI区域无有效特征");
                }
                ModelID = hv_modelID;

                // ========== 步骤3：提取模板轮廓 ==========
                // GetShapeModelContours：获取模板的形状轮廓
                // 参数1表示使用第一层金字塔（最高分辨率）
                HOperatorSet.GetShapeModelContours(out ho_modelContours, ModelID, 1);
                ModelContours = ho_modelContours;
                IsTemplateCreated = true;
            }
            catch (Exception ex)
            {
                // 创建失败时清理资源
                ho_ImageReduced?.Dispose();
                hv_modelID?.Dispose();
                ho_modelContours?.Dispose();
                ModelID?.Dispose();
                ModelID = null;
                IsTemplateCreated = false;
                throw new Exception($"创建模板失败：{ex.Message}");
            }
        }

    
        // 清除模板资源
       
        public void ClearTemplate()
        {
            ModelID?.Dispose();
            ModelID = null;
            ModelContours?.Dispose();
            ModelContours = null;
            ho_templateRegion?.Dispose();
            ho_templateRegion = null;
            ho_templateRegionXld?.Dispose();
            ho_templateRegionXld = null;
            IsTemplateCreated = false;
        }
        #endregion
    }
}
